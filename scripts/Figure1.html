Figure 1D Barplot of cell type proportions across depots

tab <- with(depots_snRNAseq@meta.data, table(Sample_Name, cellclass))

result_df <- as.data.frame(tab)
names(result_df) <- c("Sample_Name", "cellclass", "cell_count")
result_df$Tissue.type <- NA
result_df$sample <- NA
for (i in 1:nrow(result_df)) {
  result_df$Tissue.type[i] <- unique(depots_snRNAseq$Tissue.type[depots_snRNAseq$Sample_Name==result_df$Sample_Name[i]])
  result_df$sample[i] <- unique(depots_snRNAseq$sample[depots_snRNAseq$Sample_Name==result_df$Sample_Name[i]])
}

result_normalized <- result_df %>%
  group_by(Sample_Name) %>%
  mutate(percentage = (cell_count / sum(cell_count)) * 100) %>%
  ungroup()

result_normalized_immune <- result_normalized %>%
  mutate(cellclass = as.character(cellclass),
         cellclass = ifelse(cellclass %in% c("lymphoid", "myeloid"), "immune", cellclass)) %>%
  group_by(Sample_Name, Tissue.type, cellclass, sample) %>%
  summarise(cell_count = sum(cell_count),
            percentage = sum(percentage)) %>%
  ungroup()
## `summarise()` has grouped output by 'Sample_Name', 'Tissue.type', 'cellclass'.
## You can override using the `.groups` argument.
result_normalized_immune$Tissue.type[result_normalized_immune$Tissue.type=="md"] <- "ms"
result_normalized_immune$Tissue.type <- factor(result_normalized_immune$Tissue.type, levels = c("epi", "mc", "ms", "om", "sc"))


## Mesothelial 
p2 <- ggbarplot(
  data = result_normalized_immune[result_normalized_immune$cellclass == "mesothelial", ],
  x = "Tissue.type",
  y = "percentage",
  orientation = "horiz",
  fill = "#E6764F",
  add = c("mean_se", "jitter"),
  add.params = list(shape = "sample")  
) +
  scale_y_continuous(expand = c(0, 0), limits = c(-1e-3,40), breaks = c(0, 20, 40)) +

                                "4566" = 6,
                                "4567" = 10,

  NoLegend()



result_no_mesothelial <- result_normalized_immune %>%
  filter(cellclass != "mesothelial") %>%
  group_by(Sample_Name) %>%
